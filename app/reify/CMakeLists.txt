add_executable(reify main.cc)
target_link_libraries(reify libreify libpotassco)
set_target_properties(reify PROPERTIES FOLDER exe)
if(LIB_POTASSCO_BUILD_STATIC)
    if(NOT MSVC)
        target_link_options(reify PRIVATE -static-libgcc -static-libstdc++ -static)
    endif()
endif()

if(NOT CMAKE_INSTALL_BINDIR)
    message(STATUS "BINDIR not set - using bin")
    set(CMAKE_INSTALL_BINDIR "bin")
endif()
install(TARGETS reify EXPORT reify DESTINATION ${CMAKE_INSTALL_BINDIR})
